const fs=require('fs'),zlib=require('zlib');
const dir='c:/Users/Guillaume/Downloads/Scripts/LevelUp/data/cache/replays/halo_infinite/';
const files=fs.readdirSync(dir).filter(f=>f.endsWith('.json.gz'));
const res=[];
for(const f of files){
  const d=JSON.parse(zlib.gunzipSync(fs.readFileSync(dir+f)));
  const sa=d.coverage&&d.coverage.scoreActions; if(!sa||!sa.complete||/bigteam/.test(sa.group)) continue;
  const sums={};for(const a of d.scoreActions||[])sums[a.xuid]=(sums[a.xuid]||0)+a.points;
  let bad=0,tot=0,under=0,over=0,miss=0;
  for(const p of (d.scoreTimeline||{}).players||[]){const t=(p.score&&p.score.total)||[];const last=t.length?t[t.length-1].v:0;const s=sums[p.xuid]||0;tot++;if(s!==last){bad++;if(s<last){under++;miss+=last-s}else over++;}}
  if(bad>0) res.push({id:f.slice(0,8),build:d.coverage.decoder.build,group:sa.group,tot,bad,under,over,avgMiss:under?Math.round(miss/under):0,objKeys:Object.keys(d.objectives||{}).slice(0,6),rules:JSON.stringify(d.rules).slice(0,200),identity:JSON.stringify(d.identity).slice(0,300)});
}
for(const r of res) console.log(JSON.stringify(r));
console.log(res.length);
