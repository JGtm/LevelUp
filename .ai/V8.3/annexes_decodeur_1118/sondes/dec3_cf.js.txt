const fs = require('fs')
const a = JSON.parse(fs.readFileSync(process.argv[2]))
const q = (arr, p) => { const s = [...arr].sort((x, y) => x - y); return s[Math.min(s.length - 1, Math.floor(p * s.length))] }
const stat = (name, arr) => console.log(name.padEnd(26), 'n', arr.length, 'min', q(arr, 0).toFixed ? +q(arr, 0).toFixed(4) : q(arr, 0), 'p10', +(+q(arr, .1)).toFixed(4), 'p50', +(+q(arr, .5)).toFixed(4), 'p90', +(+q(arr, .9)).toFixed(4), 'max', +(+q(arr, 1)).toFixed(4))
const c = a.filter(x => x.cf && x.cf.packets)
console.log('with cf', c.length, '/', a.length)
stat('packets', c.map(x => x.cf.packets))
stat('closed/packets', c.map(x => x.cf.closed / x.cf.packets))
stat('holes/packets', c.map(x => x.cf.holes / x.cf.packets))
stat('entries', c.map(x => x.cf.entries))
stat('entries/packet', c.map(x => x.cf.entries / x.cf.packets))
stat('withAction/entries', c.map(x => x.cf.withAction / Math.max(1, x.cf.entries)))
stat('firing/entries', c.map(x => x.cf.firing / Math.max(1, x.cf.entries)))
const tot = (k) => c.reduce((s, x) => s + (x.cf[k] || 0), 0)
for (const k of ['packets', 'closed', 'holes', 'holesUnlocated', 'holesOpenViewB', 'holesOverflow', 'holesKind', 'holesBlockBC', 'holesCap', 'holesNotClosing', 'entries']) console.log(k, tot(k))
stat('entries per frame', c.map((x, i) => x.cf.entries / (a.find(y => y.f === x.f).frames)))
stat('entries/frame/roster', c.map(x => { const y = a.find(z => z.f === x.f); return x.cf.entries / y.frames / y.roster }))
