const last=s=>{if(!s||!s.total||!s.total.length)return null;return s.total[s.total.length-1].v};
const A=[],B=[];
for(const f of files){

  const b=(d.coverage.decoder&&d.coverage.decoder.build);if(b!=='HI_1_13_0')continue;
